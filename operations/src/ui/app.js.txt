const $=id=>document.getElementById(id);const sections=['overview','health','announcements','releases','controls','audit'];let announcements=[];
const fmt=new Intl.NumberFormat();const date=x=>new Date(x).toLocaleString();
function message(text,error=false){$('message').textContent=text;$('message').classList.toggle('error',error);}
async function api(path,method='GET',body){const response=await fetch('/admin/api'+path,{method,credentials:'same-origin',cache:'no-store',headers:method==='GET'?{}:{'Content-Type':'application/json'},body:body===undefined?undefined:JSON.stringify(body)});if(!response.ok){const data=await response.json().catch(()=>({}));throw new Error(data.error||'Request failed. Sign in again if your session expired.');}return response.json();}
const node=(tag,text)=>{const el=document.createElement(tag);el.textContent=text;return el;};
function button(text,action){const b=node('button',text);b.type='button';b.addEventListener('click',()=>busy(action,b));return b;}
async function busy(action,el){if(el)el.disabled=true;try{await action();}catch(e){message(e.message,true);}finally{if(el)el.disabled=false;}}
function table(target,headers,rows){const host=$(target);host.replaceChildren();if(!rows.length){host.append(node('p','No data yet.'));return;}const wrap=node('div','');wrap.className='table-wrap';const t=node('table','');const head=node('thead',''),hr=node('tr','');headers.forEach(x=>hr.append(node('th',x)));head.append(hr);t.append(head);const body=node('tbody','');for(const row of rows){const tr=node('tr','');for(const value of row){const td=node('td','');if(value instanceof Node)td.append(value);else td.textContent=String(value??'');tr.append(td);}body.append(tr);}t.append(body);wrap.append(t);host.append(wrap);}
function resetForm(){const f=$('announcement-form');f.reset();f.elements.id.value='';const local=t=>new Date(t-new Date().getTimezoneOffset()*60000).toISOString().slice(0,16);f.elements.startsAt.value=local(Date.now());f.elements.endsAt.value=local(Date.now()+7*86400000);$('editor-title').textContent='New announcement';preview();}
function preview(){const f=$('announcement-form').elements;$('preview-kind').textContent=f.kind.value.toUpperCase();$('preview-title').textContent=f.title.value||'Announcement title';$('preview-body').textContent=f.body.value||'Your message appears here.';$('preview-action').textContent=f.actionLabel.value;}
function editAnnouncement(a){const f=$('announcement-form');for(const k of ['id','title','body','kind','minCode','maxCode','actionLabel','actionUrl'])f.elements[k].value=a[k];f.elements.onceOnly.checked=a.onceOnly;for(const k of ['startsAt','endsAt'])f.elements[k].value=new Date(a[k]-new Date(a[k]).getTimezoneOffset()*60000).toISOString().slice(0,16);$('editor-title').textContent='Edit announcement';preview();f.scrollIntoView({block:'start'});}
async function loadPage(){const page=sections.includes(location.hash.slice(1))?location.hash.slice(1):'overview';sections.forEach(id=>$(id).hidden=id!==page);document.querySelectorAll('nav a').forEach(a=>{if(a.hash==='#'+page)a.setAttribute('aria-current','page');else a.removeAttribute('aria-current');});$('page-title').textContent=document.querySelector('nav a[aria-current]').textContent;message('Loading…');
 if(page==='overview'||page==='controls'){const d=await api('/overview');if(page==='overview'){$('metrics').replaceChildren();for(const [label,value]of [['Reported active today (UTC)',d.today],['Reported active · 30 days',d.active30],['First reported launches · 90 days',d.first90],['Download-link requests',d.totalClicks],['Open reports',d.openReports]]){const el=node('div','');el.className='metric';el.append(node('span',label),node('strong',fmt.format(value)));$('metrics').append(el);}table('activity',['Day (UTC)','Active installations'],d.activity.map(x=>[x.day,fmt.format(x.active)]));table('versions',['Version','Active installations'],d.versions.map(x=>[x.version,fmt.format(x.active)]));table('clicks',['Day (UTC)','Link requests'],d.clicks.map(x=>[x.day,fmt.format(x.clicks)]));}else{const f=$('controls-form').elements;f.analyticsEnabled.checked=!!d.controls.analytics_enabled;f.announcementsEnabled.checked=!!d.controls.announcements_enabled;f.reportsEnabled.checked=!!d.controls.reports_enabled;}}
 if(page==='health'){const [errors,reports]=await Promise.all([api('/health'),api('/reports')]);table('errors',['Day (UTC)','Version','Category','Count'],errors.map(x=>[x.day,x.version,x.category,x.count]));table('reports',['Submitted','App','Report','Status'],reports.map(x=>{const msg=node('div','');msg.append(node('strong',x.category),node('p',x.message));const state=node('div','');state.append(node('p',x.status));if(x.status==='open')state.append(button('Resolve',async()=>{await api('/reports/'+x.id+'/resolve','POST',{});await loadPage();}));return[date(x.created_at),x.version+' · '+x.platform,msg,state];}));}
 if(page==='announcements'){announcements=await api('/announcements');table('announcement-list',['Announcement','Display','Schedule','Actions'],announcements.map(a=>{const title=node('div','');title.append(node('strong',a.title),node('p',a.status));const actions=node('div','');actions.append(button('Edit',()=>editAnnouncement(a)));if(a.status==='published')actions.append(button('Unpublish',async()=>{await api('/announcements/'+a.id+'/unpublish','POST',{});await loadPage();}));return[title,a.kind,date(a.startsAt)+' – '+date(a.endsAt),actions];}));}
 if(page==='releases'){const d=await api('/releases');$('release-asof').textContent='GitHub totals last checked '+date(d.fetchedAt)+'. Cached for up to 15 minutes.';$('release-list').replaceChildren();for(const r of d.items){const el=node('article','');el.append(node('h2',r.tag_name),node('p',fmt.format(r.assets.reduce((n,a)=>n+a.download_count,0))+' asset downloads'));const ul=node('ul','');r.assets.filter(a=>a.name.endsWith('.apk')).forEach(a=>ul.append(node('li',a.name+' · '+fmt.format(a.download_count)+' downloads')));el.append(ul);const held=d.held.includes(r.tag_name);el.append(button(held?'Resume update offer':'Hold update offer',async()=>{if(!confirm((held?'Resume':'Hold')+' the update offer for '+r.tag_name+'?'))return;await api('/releases/hold','POST',{tag:r.tag_name,held:!held});await loadPage();}));$('release-list').append(el);}}
 if(page==='audit'){const d=await api('/audit');table('audit-list',['Time','Actor','Action','Target'],d.map(x=>[date(x.at),x.actor,x.action,x.target]));}
 message('Updated '+new Date().toLocaleTimeString());}
$('refresh').addEventListener('click',()=>busy(loadPage,$('refresh')));window.addEventListener('hashchange',()=>busy(loadPage));$('new-announcement').addEventListener('click',resetForm);$('announcement-form').addEventListener('input',preview);
$('announcement-form').addEventListener('submit',e=>{e.preventDefault();const f=e.currentTarget.elements;busy(async()=>{const data={title:f.title.value,body:f.body.value,kind:f.kind.value,status:e.submitter.value,startsAt:new Date(f.startsAt.value).getTime(),endsAt:new Date(f.endsAt.value).getTime(),minCode:Number(f.minCode.value),maxCode:Number(f.maxCode.value),actionLabel:f.actionLabel.value,actionUrl:f.actionUrl.value,onceOnly:f.onceOnly.checked};const id=f.id.value;await api('/announcements'+(id?'/'+id:''),id?'PUT':'POST',data);resetForm();await loadPage();message(data.status==='draft'?'Draft saved.':'Announcement published.');},e.submitter);});
$('controls-form').addEventListener('submit',e=>{e.preventDefault();const f=e.currentTarget.elements;busy(async()=>{await api('/controls','PUT',{analyticsEnabled:f.analyticsEnabled.checked,announcementsEnabled:f.announcementsEnabled.checked,reportsEnabled:f.reportsEnabled.checked});message('Controls saved.');},e.submitter);});resetForm();busy(loadPage);
